export const PerformerLinks = ({ items }) => {
  if (!items?.length) return null;

  return (
    <div className="SocialIcons">
      <div className="Icons">
        {items.map(({ platform, url }) =>
          url ? (
            <a
              key={`${platform}-${url}`}
              className="PlatformIcon"
              href={url}
              title={`View ${platform}`}
            >
              <span className={`icon-${platform}`} />
            </a>
          ) : null
        )}
      </div>
    </div>
  );
};

export const PerformerList = ({ performers }) => (
  <ul>
    {performers.map(({ name, links = [], website }) => (
      <li key={name} style={{ marginBottom: '0.4rem' }}>
        <span style={{ display: 'inline-flex', alignItems: 'center', gap: '0.75rem', flexWrap: 'wrap' }}>
          {website ? (
            <a href={website}>
              <strong>{name}</strong>
            </a>
          ) : (
            <strong>{name}</strong>
          )}
          {links.length > 0 && (
            <span>
              <PerformerLinks items={links} />
            </span>
          )}
        </span>
      </li>
    ))}
  </ul>
);
